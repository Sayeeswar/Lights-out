# 2) Input guardrail
#    The Realtime API normally replies the moment the caller stops talking
#    (turn_detection create_response=True). We turn that off, so nothing is
#    said until the guardrail below has judged the caller's transcript.
#    loan_input_guardrail is a real SDK InputGuardrail. RealtimeAgent has no
#    input_guardrails= option, so the SDK never runs it for us; instead we call
#    loan_input_guardrail.run(...) ourselves in handle_caller_turn().

class ScopeCheck(BaseModel):
    on_topic: bool
    reason: str
    safe_to_answer_once: bool


class CallTurnState(TypedDict):
    off_topic_answer_used: bool
    interrupted: bool
    previous_transcript: str


def _event_type_name(raw_event) -> str:
    """Return raw model event type across SDK object/dict shapes."""
    if raw_event is None:
        return ""
    if isinstance(raw_event, dict):
        return str(raw_event.get("type") or "")
    return str(getattr(raw_event, "type", "") or "")


def _event_field(raw_event, name: str, default=None):
    """Read field from both dict-like and object-like raw model events."""
    if raw_event is None:
        return default
    if isinstance(raw_event, dict):
        return raw_event.get(name, default)
    return getattr(raw_event, name, default)

@input_guardrail(run_in_parallel=False)   # run_in_parallel=False = check BEFORE the agent speaks
async def loan_input_guardrail(
    context: RunContextWrapper, agent, input: str
) -> GuardrailFunctionOutput:
    result = await Runner.run(scope_checker, input)
    check: ScopeCheck = result.final_output
    return GuardrailFunctionOutput(
        output_info=check,
        tripwire_triggered=not check.on_topic,   # True = block this question
    )


async def send_response(session, instructions: str | None = None) -> None:
    """Tell the model to reply now, optionally with one-off instructions."""
    message = {"type": "response.create"}
    if instructions:
        message["other_data"] = {"response": {"instructions": instructions}}
    await session.model.send_event(RealtimeModelSendRawMessage(message=message))


async def cancel_response(session) -> None:
    """Cancel any in-progress assistant response generation."""
    await session.model.send_event(
        RealtimeModelSendRawMessage(message={"type": "response.cancel"})
    )


async def truncate_response_item(session, item_id: str, audio_end_ms: int) -> None:
    """Trim assistant audio history at the exact heard boundary."""
    await session.model.send_event(
        RealtimeModelSendRawMessage(
            message={
                "type": "conversation.item.truncate",
                "item_id": item_id,
                "content_index": 0,
                "audio_end_ms": max(0, int(audio_end_ms)),
            }
        )
    )


def say_exactly(line: str) -> str:
    return f"Say exactly this sentence and nothing else: {line}"


async def handle_caller_turn(
    session,
    ws,
    transcript: str,
    lock: asyncio.Lock,
    turn_state: CallTurnState,
) -> None:
    """Check a caller turn, allowing one safe off-topic answer per call."""
    text = (transcript or "").strip()
    if not text:
        return  # nothing was understood, so stay quiet

    async with lock:  # keep turns in order if the caller speaks twice quickly
        continued_after_interrupt = (
            turn_state["interrupted"] and bool(turn_state["previous_transcript"])
        )
        if continued_after_interrupt:
            text = f'{turn_state["previous_transcript"]} {text}'
        turn_state["interrupted"] = False
        turn_state["previous_transcript"] = text

        try:
            result = await loan_input_guardrail.run(
                agent, text, RunContextWrapper(context=None)
            )
        except Exception as exc:  # classifier failed: do not let the turn through unchecked
            guardrail_logger.error("Input guardrail check failed: %s", exc)
            await ws.send_json({"type": "error", "text": "Could not check your question."})
            await send_response(session, say_exactly(RETRY_LINE))
            return

        if not result.output.tripwire_triggered:
            instructions = None
            if continued_after_interrupt:
                instructions = (
                    "The caller interrupted your previous answer to continue their "
                    "question. Treat the previous caller message and this continuation "
                    "as one request, then give one concise answer that addresses both."
                )
            await send_response(session, instructions)
        else:
            check: ScopeCheck = result.output.output_info
            reason = check.reason
            # Only the reason is logged, not the caller's words
            guardrail_logger.warning("Input guardrail triggered: %s", reason)
            if check.safe_to_answer_once and not turn_state["off_topic_answer_used"]:
                turn_state["off_topic_answer_used"] = True
                await ws.send_json({
                    "type": "notice",
                    "text": "I’ll answer this briefly, then return to personal-loan assistance.",
                })
                await send_response(session, OFF_TOPIC_ANSWER_INSTRUCTIONS)
            else:
                await ws.send_json({
                    "type": "notice",
                    "text": f"Off-topic question redirected. Reason: {reason}",
                })
                await send_response(session, OFF_TOPIC_REDIRECT_INSTRUCTIONS)

def extract_messages(history) -> list[dict]:
    """Turn the SDK's history items into simple {role, text} dicts for the UI."""
    messages = []
    for item in history:
        if getattr(item, "type", None) != "message":
            continue
        text = ""
        for part in getattr(item, "content", None) or []:
            text += getattr(part, "text", None) or getattr(part, "transcript", None) or ""
        text = text.strip()
        if text and text != GREETING_PROMPT:
            messages.append({"id": item.item_id, "role": item.role, "text": text})
    return messages


@app.websocket("/ws")
async def websocket_endpoint(ws: WebSocket):
    await ws.accept()
    session = await runner.run()

    async with session:
        await session.send_message(GREETING_PROMPT)  # agent speaks first

        turn_lock = asyncio.Lock()
        turn_tasks: set[asyncio.Task] = set()
        turn_state: CallTurnState = {
            "off_topic_answer_used": False,
            "interrupted": False,
            "previous_transcript": "",
        }
        pending_commit_task: asyncio.Task | None = None
        assistant_item_id: str | None = None
        assistant_played_ms = 0.0
        assistant_speaking = False

        def cancel_pending_commit() -> None:
            nonlocal pending_commit_task
            if pending_commit_task and not pending_commit_task.done():
                pending_commit_task.cancel()
            pending_commit_task = None

        async def commit_after_continuation_window() -> None:
            await asyncio.sleep(CONTINUATION_WINDOW_SECONDS)
            # If caller did not continue speaking, finalize this user turn.
            await session.model.send_event(
                RealtimeModelSendRawMessage(
                    message={"type": "input_audio_buffer.commit"}
                )
            )

        async def on_user_started_speaking() -> None:
            nonlocal assistant_speaking
            cancel_pending_commit()
            if assistant_speaking:
                turn_state["interrupted"] = True
                # Stop generation and align server conversation state with heard audio.
                await cancel_response(session)
                if assistant_item_id:
                    await truncate_response_item(
                        session, assistant_item_id, int(assistant_played_ms)
                    )
                # Tell frontend to stop local playback immediately.
                await ws.send_json({"type": "interrupted"})
                assistant_speaking = False

        async def browser_to_agent():
            """Mic audio (binary frames) from the browser -> the model."""
            while True:
                data = await ws.receive_bytes()
                await session.send_audio(data)

        async def agent_to_browser():
            """Events from the model -> the browser."""
            nonlocal pending_commit_task
            nonlocal assistant_item_id
            nonlocal assistant_played_ms
            nonlocal assistant_speaking
            async for event in session:
                if event.type == "audio":
                    await ws.send_bytes(event.audio.data)
                    assistant_speaking = True
                    assistant_played_ms += (
                        len(event.audio.data) / BYTES_PER_SAMPLE / SAMPLE_RATE_HZ
                    ) * 1000.0
                elif event.type == "audio_interrupted":
                    turn_state["interrupted"] = True
                    assistant_speaking = False
                    await ws.send_json({"type": "interrupted"})
                elif event.type == "agent_end":
                    assistant_speaking = False
                    assistant_played_ms = 0.0
                    assistant_item_id = None
                    if not turn_state["interrupted"]:
                        turn_state["previous_transcript"] = ""
                elif event.type == "history_updated":
                    await ws.send_json(
                        {"type": "history", "messages": extract_messages(event.history)}
                    )
                elif event.type == "raw_model_event":
                    data = event.data
                    raw_type = _event_type_name(data)

                    if raw_type == "input_audio_buffer.speech_started":
                        await on_user_started_speaking()
                    elif raw_type in {
                        "input_audio_buffer.speech_stopped",
                        "input_audio_buffer.timeout_triggered",
                    }:
                        # Start/restart 10-second continuation window.
                        cancel_pending_commit()
                        pending_commit_task = asyncio.create_task(
                            commit_after_continuation_window()
                        )
                    elif raw_type == "response.output_audio.delta":
                        # Track which assistant item is being spoken for truncation.
                        item_id = _event_field(data, "item_id")
                        if isinstance(item_id, str) and item_id:
                            assistant_item_id = item_id
                    elif raw_type == "input_audio_transcription_completed":
                        # Run the check in the background so audio keeps flowing
                        task = asyncio.create_task(
                            handle_caller_turn(
                                session, ws, data.transcript, turn_lock, turn_state
                            )
                        )
                        turn_tasks.add(task)
                        task.add_done_callback(turn_tasks.discard)
                elif event.type == "error":
                    await ws.send_json({"type": "error", "text": str(event.error)})

        tasks = [
            asyncio.create_task(browser_to_agent()),
            asyncio.create_task(agent_to_browser()),
        ]
        try:
            await asyncio.wait(tasks, return_when=asyncio.FIRST_COMPLETED)
        except WebSocketDisconnect:
            pass
        finally:
            cancel_pending_commit()
            for t in tasks + list(turn_tasks):
                t.cancel()
